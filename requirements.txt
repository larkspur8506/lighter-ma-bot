# Runtime dependencies — pinned/constrained.
#
# Reproducibility rule: the web stack is pinned to exact versions on purpose.
# `tests/test_requirements.py` asserts that every line here is installed and
# that the installed version satisfies the constraint, so a silently-changed
# resolution fails the build instead of surfacing later as a mysterious error.

# lighter-sdk 1.1.8 ships the WASM signer binaries inside the wheel
# (lighter/signers/lighter-signer-linux-amd64.so et al.), so the Docker
# base image must be linux/amd64.
lighter-sdk==1.1.8

# HTTP service (monitoring panel + JSON API, T10).
# Pinned exactly: starlette 1.10 moved its TestClient to `httpx2` and only
# falls back to `httpx` with a StarletteDeprecationWarning. The trio below is
# the combination that was actually resolved, installed and verified.
fastapi==0.162.0
starlette==1.10.0
uvicorn==0.54.0

# TestClient transport. `httpx2` is what starlette 1.10 imports first; `httpx`
# is the legacy fallback and is capped at <0.29 because that is the range
# starlette 1.10's own `[full]` extra declares (0.29+ is untested against it).
# Both are declared so neither path can disappear silently.
httpx2>=2.0
httpx>=0.27,<0.29

python-multipart>=0.0.18,<1.0

# apscheduler drives the 4h bar tick
apscheduler>=3.10,<4.0

# Used by tests/test_requirements.py to parse the constraints in this file.
packaging>=23.2,<28
